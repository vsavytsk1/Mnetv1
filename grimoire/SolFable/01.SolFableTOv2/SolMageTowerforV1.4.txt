```python
    """Return (F_n, F_{n+1}) by exact fast doubling."""
    if n == 0:
        return 0, 1

    a, b = fib_pair(n >> 1)
    c = a * ((b << 1) - a)

def exact_triangulation(n: int) -> int:
```

```python
def rank_mod(matrix: np.ndarray, prime: int) -> int:
    """Exact Gaussian rank over F_prime."""
    a = np.mod(matrix, prime).astype(np.int64, copy=True)
    rows, cols = a.shape
    rank = 0

    for col in range(cols):
        candidates = np.flatnonzero(a[rank:, col])

        if candidates.size == 0:
            continue

        pivot = rank + int(candidates[0])

        if pivot != rank:
            a[[rank, pivot]] = a[[pivot, rank]]

        inverse = pow(int(a[rank, col]), -1, prime)
        a[rank, col:] = (a[rank, col:] * inverse) % prime

        if rank + 1 < rows:
            factors = a[rank + 1:, col].copy()
            mask = factors != 0

            if np.any(mask):
                target_rows = np.where(mask)[0] + rank + 1
                a[target_rows, col:] = (
                    a[target_rows, col:]
                    - factors[mask, None] * a[rank, col:]
                ) % prime

        rank += 1

        if rank == min(rows, cols):
            break

    return rank
```

```python
    dirac_basis: np.ndarray,
    left_action: np.ndarray,
    opposite_action: np.ndarray,
) -> np.ndarray:
    first_commutator = (
        dirac_basis @ left_action
        - left_action @ dirac_basis
```
